like the shift branches', since their ordinary lines announce the resulting
value (already actual) rather than a delta.

Effects are applied via the Django shell (`make shell` from the worktree).
Paste this prelude first, setting your character's name:

    from datetime import timedelta
    from django.utils import timezone
    from apps.shyland.models import (Character, EffectDefinition,
        EffectComponent, EffectInstance, EffectComponentInstance)

    CHAR = Character.objects.get(name__iexact='YOUR-CHARACTER-NAME')

    def apply_tick_effect(ctype, magnitude, minutes=10):
        d, _ = EffectDefinition.objects.get_or_create(
            slug=f"playtest-{ctype.replace('_', '-')}",
            defaults={'name': f'Playtest {ctype}'})
        c, _ = EffectComponent.objects.get_or_create(
            definition=d, component_type=ctype, order=0,
            defaults={'magnitude_base': magnitude, 'magnitude_scaling': 0.0,
                      'duration_base': 600.0, 'duration_scaling': 0.0})
        ei = EffectInstance.objects.create(
            definition=d, target=CHAR, mk_tier=1, is_active=True)
        EffectComponentInstance.objects.create(
            effect_instance=ei, component=c, magnitude=magnitude,
            is_active=True,
            expires_at=timezone.now() + timedelta(minutes=minutes))
        return ei
